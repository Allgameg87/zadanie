Задание 3: Программа для взаимодействия с API VirusTotal

Python скрипт для работы с VirusTotal API:

import requests
import json
import time

API_KEY = 'ваш_api_ключ_virustotal'
BASE_URL = 'https://www.virustotal.com/api/v3/'

def scan_file(file_path):
    """Отправка файла на анализ в VirusTotal"""
    url = BASE_URL + 'files'
    headers = {'x-apikey': API_KEY}
    
    with open(file_path, 'rb') as file:
        files = {'file': (file_path, file)}
        response = requests.post(url, headers=headers, files=files)
    
    if response.status_code == 200:
        result = response.json()
        return result['data']['id']  # Возвращает ID анализа
    else:
        print(f"Ошибка при отправке файла: {response.status_code}")
        return None

def get_report(analysis_id):
    """Получение отчета по анализу"""
    url = BASE_URL + 'analyses/' + analysis_id
    headers = {'x-apikey': API_KEY}
    
    response = requests.get(url, headers=headers)
    
    if response.status_code == 200:
        return response.json()
    else:
        print(f"Ошибка при получении отчета: {response.status_code}")
        return None

def check_url(url_to_check):
    """Проверка URL через VirusTotal"""
    url = BASE_URL + 'urls'
    headers = {
        'x-apikey': API_KEY,
        'Content-Type': 'application/x-www-form-urlencoded'
    }
    
    response = requests.post(url, headers=headers, data={'url': url_to_check})
    
    if response.status_code == 200:
        result = response.json()
        url_id = result['data']['id']
        # Получаем полный отчет
        report_url = BASE_URL + 'urls/' + url_id
        report_response = requests.get(report_url, headers=headers)
        return report_response.json()
    else:
        print(f"Ошибка при проверке URL: {response.status_code}")
        return None

# Пример использования
if __name__ == "__main__":
    print("1. Сканировать файл")
    print("2. Проверить URL")
    choice = input("Выберите действие (1/2): ")
    
    if choice == '1':
        file_path = input("Введите путь к файлу: ")
        analysis_id = scan_file(file_path)
        if analysis_id:
            print("Файл отправлен на анализ. Ожидание отчета...")
            time.sleep(30)  # Даем время на анализ
            report = get_report(analysis_id)
            print("Результаты анализа:")
            print(json.dumps(report, indent=2))
    elif choice == '2':
        url = input("Введите URL для проверки: ")
        result = check_url(url)
        if result:
            print("Результаты проверки URL:")
            print(json.dumps(result, indent=2))
    else:
        print("Неверный выбор")

Требования для работы:

1 Зарегистрироваться на VirusTotal и получить API ключ

2 Установить библиотеку requests: pip install requests

3 Вставить полученный API ключ в переменную API_KEY в коде
